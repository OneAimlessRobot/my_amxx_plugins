    Get_BitVar(g_Had_Arifle, thing_goes_here)

    Find: Get_BitVar\(g_Had_Arifle, ([a-zA-Z_]*)\)

    Replace: sh_get_id_custom_weapon_bit($1, SH_ASSAULT_RIFLE)

    Find: UnSet_BitVar\(g_Had_Arifle, ([a-zA-Z_]*) \)

    Replace: sh_assign_id_custom_weapon_bit($1, SH_ASSAULT_RIFLE, false)


    Find: Set_BitVar\(g_Had_Arifle, ([a-zA-Z_]*) \)

    Replace: sh_assign_id_custom_weapon_bit($1, SH_ASSAULT_RIFLE, true)

        ([a-zA-Z_]*) -> corresponds to:
        
        "Any number of alphanumeric characters here. turn it into a token"
